
// --- scenery ---------------------------------------------------------------------------------------------------
function branch(c, x, y, ang, len, wd, col, d, tm) {
  if (d > 3 || len < 4) return;
  const x2 = x + Math.cos(ang) * len, y2 = y + Math.sin(ang) * len;
  c.lineCap = 'round'; c.strokeStyle = col; c.lineWidth = wd; c.beginPath(); c.moveTo(x, y); c.lineTo(x2, y2); c.stroke();
  const sw = Math.sin(tm * 1.2 + d) * 0.04;
  branch(c, x2, y2, ang - 0.5 + sw, len * 0.72, wd * 0.72, col, d + 1, tm); branch(c, x2, y2, ang + 0.5 + sw, len * 0.72, wd * 0.72, col, d + 1, tm);
}
function anemone(c, x, y, s, tm) {
  c.save(); c.translate(x, y);
  c.fillStyle = lg(c, 0, -s * 0.3, 0, 0, [[0, '#c04a9a'], [1, '#7a2a6a']]); c.beginPath(); c.roundRect(-s * 0.32, -s * 0.34, s * 0.64, s * 0.36, s * 0.12); c.fill();
  for (let i = 0; i < 11; i++) {
    const f = (i - 5) / 5, bx = f * s * 0.28, sw = Math.sin(tm * 1.8 + i * 0.8) * s * 0.07;
    c.strokeStyle = i % 2 ? '#f08ac8' : '#e86ab4'; c.lineWidth = Math.max(2, s * 0.07); c.lineCap = 'round';
    c.beginPath(); c.moveTo(bx, -s * 0.3); c.quadraticCurveTo(bx + f * s * 0.3 + sw, -s * 0.62, bx + f * s * 0.5 + sw * 1.6, -s * 0.92); c.stroke();
    c.fillStyle = '#ffd0ec'; circ(c, bx + f * s * 0.5 + sw * 1.6, -s * 0.92, s * 0.06); c.fill();
  }
  c.restore();
}
function drawSea(c, w, seaH, surfY, tm) {
  const zw = w / 4, wh = seaH - surfY;
  c.fillStyle = lg(c, 0, 0, w, 0, [[0, '#ffe1a6'], [0.3, '#a8defa'], [1, '#d7edf9']]); c.fillRect(0, 0, w, surfY + 4);
  c.fillStyle = rg(c, w * 0.09, surfY * 0.45, 0, surfY * 1.5, [[0, 'rgba(255,250,210,1)'], [0.3, 'rgba(255,232,150,.75)'], [1, 'rgba(255,220,120,0)']]); c.fillRect(0, 0, w * 0.45, surfY + 4);
  c.fillStyle = lg(c, 0, 0, w, 0, [[0, '#38d6cb'], [0.2, '#2fc6bd'], [0.32, '#2aa88e'], [0.44, '#26937f'], [0.62, '#2f78d6'], [0.76, '#3a86dc'], [0.88, '#8fd0ea'], [1, '#a7dcf0']]); c.fillRect(0, surfY, w, wh);
  c.fillStyle = lg(c, 0, surfY, 0, seaH, [[0, 'rgba(255,255,255,.14)'], [0.35, 'rgba(0,30,80,.05)'], [1, 'rgba(2,14,48,.55)']]); c.fillRect(0, surfY, w, wh);
  // sunbeams from the top left
  c.save(); c.beginPath(); c.rect(0, surfY, w, wh); c.clip();
  for (let i = 0; i < 5; i++) {
    const x = w * (0.02 + i * 0.2) + Math.sin(tm * 0.35 + i) * 8;
    c.fillStyle = lg(c, x, surfY, x + 90, seaH, [[0, 'rgba(255,255,255,.14)'], [1, 'rgba(255,255,255,0)']]);
    c.beginPath(); c.moveTo(x, surfY); c.lineTo(x + 34, surfY); c.lineTo(x + 34 + 110, seaH); c.lineTo(x + 100, seaH); c.closePath(); c.fill();
  }
  c.restore();
  // sea floor
  const fl = (x) => seaH - 26 + Math.sin(x * 0.035 + 1) * 5 + Math.sin(x * 0.09) * 2;
  c.beginPath(); c.moveTo(0, seaH); for (let x = 0; x <= w + 6; x += 6) c.lineTo(x, fl(x)); c.lineTo(w, seaH); c.closePath();
  c.fillStyle = lg(c, 0, 0, w, 0, [[0, '#f0dca4'], [0.2, '#e4cd94'], [0.3, '#9a8a64'], [0.44, '#7c6e50'], [0.6, '#27386e'], [0.76, '#2b3d78'], [0.86, '#c8deec'], [1, '#d6e8f2']]); c.fill();
  c.fillStyle = lg(c, 0, seaH - 34, 0, seaH, [[0, 'rgba(255,255,255,.2)'], [1, 'rgba(0,0,0,.3)']]); c.fill();
  // reef: corals and an anemone
  c.save(); c.beginPath(); c.rect(0, surfY, zw, seaH - surfY); c.clip();
  const gy = (f) => fl(zw * f);
  shadow(c, zw * 0.2, gy(0.2) + 2, zw * 0.12, 4); branch(c, zw * 0.2, gy(0.2), -Math.PI / 2, wh * 0.13, 6, '#ff8a9a', 0, tm);
  shadow(c, zw * 0.82, gy(0.82) + 2, zw * 0.1, 4); branch(c, zw * 0.82, gy(0.82), -Math.PI / 2 - 0.1, wh * 0.1, 5, '#ffa04a', 0, tm);
  const dm = zw * 0.17, dx = zw * 0.52, dy = gy(0.52) + 2; shadow(c, dx, dy + 2, dm * 1.1, 4);
  c.fillStyle = lg(c, dx - dm, dy - dm, dx + dm, dy, [[0, '#ffe38a'], [1, '#e0a030']]); c.beginPath(); c.ellipse(dx, dy, dm, dm * 0.75, 0, Math.PI, TAU); c.closePath(); c.fill();
  c.strokeStyle = 'rgba(150,90,10,.45)'; c.lineWidth = 1.5; for (let i = 1; i < 3; i++) { c.beginPath(); c.ellipse(dx, dy, dm * i / 3, dm * 0.75 * i / 3, 0, Math.PI, TAU); c.stroke(); }
  anemone(c, zw * 0.36, gy(0.36) + 3, zw * 0.46, tm);
  c.restore();
  // kelp forest
  c.save(); c.beginPath(); c.rect(zw, surfY, zw, seaH - surfY); c.clip();
  for (let i = 0; i < 5; i++) {
    const bx = zw * (1.1 + i * 0.19), top = surfY + 16 + (i % 3) * 14, ph = i * 1.3;
    const pts = []; for (let y = fl(bx); y > top; y -= 9) pts.push([bx + Math.sin(y * 0.045 + tm * 1.1 + ph) * (6 + (fl(bx) - y) * 0.03), y]);
    c.lineCap = 'round'; c.lineJoin = 'round'; c.strokeStyle = i % 2 ? '#2e8a3c' : '#3fa94b'; c.lineWidth = 6;
    c.beginPath(); pts.forEach((p, k) => (k ? c.lineTo(p[0], p[1]) : c.moveTo(p[0], p[1]))); c.stroke();
    c.fillStyle = i % 2 ? '#3c9a45' : '#52c05a';
    pts.forEach((p, k) => { if (k % 3 === 1) { ell(c, p[0] + (k % 2 ? 7 : -7), p[1], 8, 3, k % 2 ? -0.5 : 0.5); c.fill(); } });
    const tp = pts[pts.length - 1]; c.fillStyle = '#6ad06a'; ell(c, tp[0], tp[1], 5, 9, 0.2); c.fill();
    c.fillStyle = '#6d6a62'; ell(c, bx, fl(bx) + 2, 11, 6); c.fill();
  }
  c.fillStyle = '#7a3a8a'; for (let k = 0; k < 10; k++) { const a = (k / 10) * TAU; c.strokeStyle = '#a65cb5'; c.lineWidth = 1.5; c.beginPath(); c.moveTo(zw * 1.52 + Math.cos(a) * 6, fl(zw * 1.52) - 5 + Math.sin(a) * 6); c.lineTo(zw * 1.52 + Math.cos(a) * 11, fl(zw * 1.52) - 5 + Math.sin(a) * 11); c.stroke(); }
  circ(c, zw * 1.52, fl(zw * 1.52) - 5, 6.5); c.fill();
  c.restore();
  // open sea: a far school and bubbles
  c.save(); c.beginPath(); c.rect(zw * 2, surfY, zw, seaH - surfY); c.clip();
  const rr = rnd(7);
  for (let i = 0; i < 9; i++) {
    const px = zw * 2 + ((rr() * zw + tm * (8 + i)) % zw), py = surfY + wh * (0.3 + rr() * 0.4) + Math.sin(tm + i) * 4;
    c.fillStyle = 'rgba(196,226,255,.55)'; ell(c, px, py, 6, 2.6); c.fill(); c.beginPath(); c.moveTo(px - 5, py); c.lineTo(px - 10, py - 3.5); c.lineTo(px - 10, py + 3.5); c.fill();
  }
  c.fillStyle = 'rgba(255,255,255,.35)'; for (let i = 0; i < 7; i++) { const py = seaH - ((tm * 18 + i * 53) % wh); circ(c, zw * 2 + zw * (0.15 + 0.12 * i) % zw, py, 1.5 + (i % 3)); c.fill(); }
  c.restore();
  // icy water: a floe, chunks under the surface
  c.save(); c.beginPath(); c.rect(zw * 3, 0, zw, seaH); c.clip();
  const fx = zw * 3.5 + Math.sin(tm * 0.5) * 3, fw = zw * 0.82, uy = surfY + wh * 0.26;
  c.fillStyle = lg(c, 0, surfY, 0, uy, [[0, 'rgba(225,246,255,.6)'], [1, 'rgba(180,225,245,.08)']]);
  c.beginPath(); c.moveTo(fx - fw / 2, surfY); c.lineTo(fx + fw / 2, surfY); c.lineTo(fx + fw * 0.36, surfY + wh * 0.12); c.lineTo(fx + fw * 0.2, uy); c.lineTo(fx + fw * 0.05, surfY + wh * 0.16); c.lineTo(fx - fw * 0.14, uy * 1.02); c.lineTo(fx - fw * 0.34, surfY + wh * 0.1); c.closePath(); c.fill();
  shadow(c, fx, surfY + 3, fw * 0.55, 5, 0.2);
  c.fillStyle = lg(c, fx - fw / 2, surfY - 20, fx + fw / 2, surfY, [[0, '#ffffff'], [0.7, '#dff1fa'], [1, '#b6d9ec']]);
  c.beginPath(); c.moveTo(fx - fw / 2, surfY + 4); c.quadraticCurveTo(fx - fw * 0.5, surfY - 14, fx - fw * 0.3, surfY - 17); c.lineTo(fx + fw * 0.1, surfY - 19); c.quadraticCurveTo(fx + fw * 0.45, surfY - 14, fx + fw / 2, surfY + 4); c.closePath(); c.fill();
  c.strokeStyle = 'rgba(120,170,200,.7)'; c.lineWidth = 1.5; c.stroke();
  c.fillStyle = 'rgba(200,235,250,.35)'; for (const [f, y, r2] of [[0.2, 0.55, 6], [0.7, 0.65, 8], [0.45, 0.82, 5]]) { ell(c, zw * 3 + zw * f, surfY + wh * y, r2 * 1.5, r2); c.fill(); }
  c.fillStyle = 'rgba(255,255,255,.8)'; for (let i = 0; i < 6; i++) { const a = 0.5 + 0.5 * Math.sin(tm * 2 + i * 1.7); circ(c, zw * 3 + zw * (0.1 + 0.15 * i), surfY + wh * (0.2 + 0.1 * ((i * 3) % 5)), 1 + a * 1.3); c.fill(); }
  c.restore();
  // waves on the surface
  c.beginPath(); c.moveTo(0, surfY); for (let x = 0; x <= w + 6; x += 6) c.lineTo(x, surfY + Math.sin(x * 0.05 + tm * 1.5) * 2.2); c.strokeStyle = 'rgba(255,255,255,.65)'; c.lineWidth = 2; c.stroke();
}
