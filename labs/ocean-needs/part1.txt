function wrapLines(c, text, maxW) {
  const words = String(text).split(' '), lines = []; let cur = '';
  for (const w of words) {
    const t = cur ? cur + ' ' + w : w;
    if (c.measureText(t).width > maxW && cur) { lines.push(cur); cur = w; } else cur = t;
  }
  if (cur) lines.push(cur);
  return lines;
}
// text on a dark rounded pill, wrapped to the canvas and clamped inside it; returns its height
function pill(c, text, cx, y, W, o = {}) {
  const size = o.size || 12, maxW = Math.min(o.maxW || W - 12, W - 12);
  c.save(); c.font = `800 ${size}px Nunito, sans-serif`;
  const lines = wrapLines(c, text, maxW - 14), tw = Math.max(...lines.map((l) => c.measureText(l).width));
  const bw = Math.min(W - 8, tw + 14), bh = lines.length * (size + 3) + 8, x = clamp(cx - bw / 2, 4, Math.max(4, W - bw - 4));
  c.fillStyle = o.bg || 'rgba(8,14,32,.82)'; c.beginPath(); c.roundRect(x, y, bw, bh, Math.min(13, bh / 2)); c.fill();
  c.fillStyle = o.color || '#fff'; c.textAlign = 'center'; c.textBaseline = 'middle';
  lines.forEach((l, i) => c.fillText(l, x + bw / 2, y + 4 + (size + 3) * (i + 0.5) + 1));
  c.restore();
  return bh;
}

}
// outline first (wide), then fills: gives one clean outline around overlapping shapes
function blob(c, paths, fill, stroke, lw) {
  c.save(); c.lineJoin = 'round'; c.strokeStyle = stroke; c.lineWidth = lw * 2;
  paths.forEach((p) => c.stroke(p));
  c.fillStyle = fill; paths.forEach((p) => c.fill(p));
  c.restore();
}

function poly(c, pts) {
  let a = 0;
  for (let i = 0; i < pts.length; i++) { const p = pts[i], q = pts[(i + 1) % pts.length]; a += p[0] * q[1] - q[0] * p[1]; }
  const P = a < 0 ? pts.slice().reverse() : pts;
  c.moveTo(P[0][0], P[0][1]);
  for (let i = 1; i < P.length; i++) c.lineTo(P[i][0], P[i][1]);
  c.closePath();
}
