// the octopus: head and eight tentacles as one outline so a clip can follow it
function octoPath(tm, curl = 1) {
  const c = new Path2D();
  const hp = []; for (let i = 0; i < 40; i++) { const a = (i / 40) * TAU; hp.push([Math.cos(a) * 0.52, -0.32 + Math.sin(a) * 0.6]); }
  poly(c, hp);
  for (let k = 0; k < 8; k++) {
    const f = (k - 3.5) / 3.5, bx = f * 0.36, A = [], B = [], N = 14;
    for (let i = 0; i <= N; i++) {
      const u = i / N, sway = Math.sin(tm * 2 + k * 0.9 - u * 3.2) * 0.09 * u * curl;
      const cx = bx + f * 0.45 * u + Math.sin(u * 3.4 + k) * 0.12 * u * curl + sway, cy = 0.1 + u * 0.95, wd = 0.115 * (1 - u * 0.88) + 0.012;
      A.push([cx - wd, cy]); B.push([cx + wd, cy]);
    }
    poly(c, A.concat(B.reverse()));
  }
  return c;
}
function octopus(c, tm, o = {}) {
  const col = o.col || '#b45fd6', bl = o.blend || 0;
  const OP = octoPath(tm, o.curl ?? 1);
  c.fillStyle = lg(c, -0.4, -0.9, 0.5, 0.6, [[0, shade(col, 0.22)], [0.6, col], [1, shade(col, -0.28)]]); c.fill(OP);
  c.save(); c.clip(OP);
  if (o.paint) o.paint(c);
  c.fillStyle = `rgba(255,255,255,${0.22 * (1 - bl)})`; ell(c, -0.18, -0.62, 0.22, 0.12, -0.6); c.fill();
  if (bl < 0.8) {
    c.fillStyle = `rgba(255,255,255,${0.3 * (1 - bl)})`;
    for (let k = 0; k < 8; k++) { const f = (k - 3.5) / 3.5; for (let i = 3; i < 11; i += 2) { const u = i / 12; circ(c, f * 0.36 + f * 0.45 * u + Math.sin(u * 3.4 + k) * 0.12 * u * (o.curl ?? 1), 0.1 + u * 0.95, 0.03 * (1 - u * 0.5)); c.fill(); } }
  }
  c.restore();
  c.lineJoin = 'round'; c.lineWidth = 0.035; c.strokeStyle = `rgba(${bl > 0.5 ? '90,70,40' : '60,20,90'},${0.75 - 0.5 * bl})`; c.stroke(OP);
  const ea = 1 - 0.55 * bl, wide = o.wide ? 1.2 : 1;
  c.save(); c.globalAlpha = ea;
  for (const sx of [-0.2, 0.2]) { eye(c, sx, -0.18, 0.13 * wide, { white: true }); }
  c.fillStyle = 'rgba(255,120,140,.35)'; circ(c, -0.36, -0.02, 0.07); c.fill(); circ(c, 0.36, -0.02, 0.07); c.fill();
  c.strokeStyle = '#4a1f66'; c.lineWidth = 0.03; c.beginPath(); c.moveTo(-0.08, 0.04); c.quadraticCurveTo(0, o.wide ? 0.0 : 0.1, 0.08, 0.04); c.stroke();
  c.restore();
}
