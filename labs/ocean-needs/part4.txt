let G = null; // geometry for the pointer
const zoneOf = (x) => clamp(Math.floor(x / (G.w / 4)), 0, 3);
function drawHome(c, w, h, tm) {
  const t = T[lang], H = S.h, trayH = Math.min(104, h * 0.21), seaH = h - trayH, surfY = Math.round(seaH * 0.13), zw = w / 4;
  G = { w, h, seaH, trayH, surfY };
  drawSea(c, w, seaH, surfY, tm);
  c.strokeStyle = 'rgba(255,255,255,.2)'; c.lineWidth = 1.5; c.setLineDash([4, 6]);
  for (let i = 1; i < 4; i++) { c.beginPath(); c.moveTo(zw * i, surfY + 2); c.lineTo(zw * i, seaH); c.stroke(); }
  c.setLineDash([]);
  const dz = H.drag && H.drag.moved && H.drag.y < seaH ? zoneOf(H.drag.x) : -1;
  if (dz >= 0) { c.fillStyle = 'rgba(255,255,255,.16)'; c.fillRect(zw * dz, surfY, zw, seaH - surfY); }
  const pl = H.placed, el = pl ? tm - pl.t0 : 0;
  if (pl) {
    const pulse = 0.65 + 0.35 * Math.sin(tm * 3);
    for (const z of bestZones(pl.a)) {
      c.strokeStyle = `rgba(61,220,151,${easeOut(el / 0.6) * pulse})`; c.lineWidth = 3; c.beginPath(); c.roundRect(zw * z + 3, surfY + 3, zw - 6, seaH - surfY - 6, 10); c.stroke();
    }
  }
  for (let i = 0; i < 4; i++) pill(c, t.zones[i], zw * (i + 0.5), surfY + 8, w, { maxW: zw - 4 });
  if (pl) {
    for (const z of bestZones(pl.a)) pill(c, t.best, zw * (z + 0.5), seaH - 44, w, { maxW: zw - 4, bg: 'rgba(14,90,60,.88)', color: '#d6ffe9' });
    // the animal in its place
    const cx = zw * (pl.z + 0.5) + Math.sin(tm * 1.3) * zw * 0.08, cy = surfY + (seaH - surfY) * 0.68 + Math.sin(tm * 1.7) * 5;
    const s = Math.min(zw * 0.3, 34), pop = 1 + 0.35 * Math.max(0, 1 - el * 3) * Math.cos(el * 14);
    shadow(c, cx, seaH - 18, s * 1.1, 4, 0.22);
    animalAt(c, pl.a, cx, cy, s * pop, tm, { rot: Math.sin(tm * 1.3) * 0.06 });
    drawCard(c, w, surfY + 52, pl, el);
  } else {
    pill(c, H.sel ? t.home.tap : t.home.hint, w / 2, surfY + 52, w, { maxW: Math.min(w - 20, 300), size: 13 });
  }
  // tray
  c.fillStyle = lg(c, 0, seaH, 0, h, [[0, '#101a36'], [1, '#0a1128']]); c.fillRect(0, seaH, w, trayH);
  c.fillStyle = 'rgba(130,145,190,.3)'; c.fillRect(0, seaH, w, 1.5);
  const sw = w / 6;
  KEYS.forEach((k, i) => {
    const x = sw * i + 3, y = seaH + 6, bw = sw - 6, bh = trayH - 12, sel = H.sel === k, used = pl && pl.a === k;
    c.fillStyle = sel ? 'rgba(255,255,255,.2)' : 'rgba(130,145,190,.12)'; c.beginPath(); c.roundRect(x, y, bw, bh, 12); c.fill();
    c.strokeStyle = sel ? '#fff' : 'rgba(130,145,190,.3)'; c.lineWidth = sel ? 2 : 1; c.stroke();
    c.save(); c.globalAlpha = used ? 0.3 : 1;
    animalAt(c, k, x + bw / 2, y + bh * 0.4 + Math.sin(tm * 2 + i) * 1.5, Math.min(bw * 0.36, 24), tm);
    c.restore();
    c.fillStyle = '#d4dcf0'; c.font = '800 12px Nunito, sans-serif'; c.textAlign = 'center'; c.textBaseline = 'middle'; c.fillText(t.an[k], x + bw / 2, y + bh - 11);
  });
  if (H.drag && H.drag.moved) {
    const d = H.drag; shadow(c, d.x, d.y + 22, 26, 6, 0.3);
    animalAt(c, d.key, d.x, d.y - 36, 34, tm, { rot: Math.sin(tm * 8) * 0.08 });
  }
}
function drawCard(c, w, y, pl, el) {
  const t = T[lang], m = MX[pl.a][pl.z], n = sum3(m), why = whyText(pl.a, pl.z);
  c.save(); c.font = '700 12px Nunito, sans-serif';
  const cw = Math.min(w - 16, 380), cx = (w - cw) / 2, lines = wrapLines(c, why, cw - 24), ch = 30 + 3 * 24 + 8 + lines.length * 15 + 10;
  c.fillStyle = 'rgba(8,14,34,.86)'; c.beginPath(); c.roundRect(cx, y, cw, ch, 14); c.fill();
  c.strokeStyle = 'rgba(160,180,230,.35)'; c.lineWidth = 1; c.stroke();
  c.fillStyle = '#fff'; c.font = '900 14px Nunito, sans-serif'; c.textAlign = 'left'; c.textBaseline = 'middle';
  c.fillText(`${t.ful[pl.a]}`, cx + 12, y + 16);
  c.textAlign = 'right'; c.fillStyle = n === 3 ? '#7dffc0' : '#ffd08a'; c.fillText(t.home.of(n), cx + cw - 12, y + 16);
  for (let i = 0; i < 3; i++) {
    const ry = y + 30 + i * 24 + 12, p = easeOut((el - 0.25 - i * 0.22) / 0.6), v = (m[i] ? 1 : 0.22) * p;
    c.textAlign = 'left'; c.font = '800 12px Nunito, sans-serif'; c.fillStyle = '#dbe4fb';
    c.fillText(`${['🍽️', '🏠', '🌡️'][i]} ${t.needs[i]}`, cx + 12, ry);
    const bx = cx + 12 + Math.min(110, cw * 0.38), bw = cx + cw - 36 - bx;
    c.fillStyle = 'rgba(130,145,190,.22)'; c.beginPath(); c.roundRect(bx, ry - 6, bw, 12, 6); c.fill();

function flow(c, x, y, sx, sy, rough, tm) {
  c.save(); c.lineWidth = 2; c.strokeStyle = 'rgba(255,255,255,.4)'; c.setLineDash([10, 9]); c.lineDashOffset = tm * 90; c.lineCap = 'round';
  for (const k of [-1, 1]) for (const m of [1.5, 2.1]) {
    const yy = y + k * sy * m;
    c.beginPath(); c.moveTo(x - sx * 2.4, yy);
    c.bezierCurveTo(x - sx * 1.2, yy, x - sx * (rough ? 1.0 : 0.6), y + k * sy * m * (rough ? 1.25 : 1.1), x, y + k * sy * m * (rough ? 1.2 : 1.05));
    c.bezierCurveTo(x + sx * 0.7, y + k * sy * m * 1.05, x + sx * 1.4, yy + (rough ? k * sy * 0.9 : 0), x + sx * 2.4, yy + (rough ? k * sy * 0.6 : 0)); c.stroke();
  }
  c.restore();
}

const lerp2 = (a, b, t) => a + (b - a) * t;
